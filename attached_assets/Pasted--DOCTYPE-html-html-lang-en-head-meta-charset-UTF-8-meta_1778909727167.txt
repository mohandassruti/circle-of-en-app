<script>
  function setHomeState(state) {
    const findLink    = document.getElementById('find-link');
    const navCount    = document.getElementById('home-nav-count');
    const statGiven   = document.getElementById('stat-given');
    const statReceived= document.getElementById('stat-received');

    if (state === 'returning') {
      findLink.style.display = 'block';
      navCount.textContent   = '4 given · 2 received';
      statGiven.textContent  = '4';
      statReceived.textContent = '2';
    } else {
      findLink.style.display = 'none';
      navCount.textContent   = '';
      statGiven.textContent  = '0';
      statReceived.textContent = '0';
    }
  }

  // Default to first-visit state on load
  setHomeState('first');

  function show(id) {
    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    document.getElementById(id).classList.add('active');
    window.scrollTo(0,0);
    const map = {
      'screen-home':0,'screen-give':1,'screen-given':2,
      'screen-receive-tag':3,'screen-en':4,'screen-limit':5
    };
    const btns = document.querySelectorAll('.proto-btn');
    btns.forEach(b => b.classList.remove('active'));
    if (map[id] !== undefined) btns[map[id]].classList.add('active');
  }

  function setActive(el) {
    document.querySelectorAll('.proto-btn').forEach(b => b.classList.remove('active'));
    el.classList.add('active');
  }

  function toggleTag(el) { el.classList.toggle('active'); }

  function toggleReceiveTag(el) {
    document.querySelectorAll('#receive-tags .tag-chip').forEach(t => t.classList.remove('active'));
    el.classList.add('active');
  }

  function updateCount() {
    const ta = document.getElementById('give-text');
    document.getElementById('char-count').textContent = (400 - ta.value.length) + ' characters left';
  }

  function submitGive() {
    const text = document.getElementById('give-text').value.trim();
    if (!text) {
      document.getElementById('give-text').focus();
      return;
    }
    show('screen-given');
    document.getElementById('give-text').value = '';
    document.querySelectorAll('#give-tags .tag-chip').forEach(t => t.classList.remove('active'));
    updateCount();
  }

  ];
